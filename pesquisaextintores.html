function pesquisarExtintor(){ let valor = document .getElementById("pesquisa") .value .toLowerCase(); let linhas = document.querySelectorAll("#tabelaExtintores tr"); linhas.forEach(linha => { let textoLinha = linha.innerText.toLowerCase(); if(textoLinha.includes(valor)){ linha.style.display = ""; }else{ linha.style.display = "none"; } }); }
